####### x86 #######
###################

##### FFT #####

build/X86/gem5.opt --outdir=m5out/weak/x86/FFT configs/example/se.py \
--cpu-type=X86MinorCPU \
--cpu-clock=1GHz \
--cacheline_size=64 \
--caches \
--l1i_size=8kB \
--l1i_assoc=4 \
--l1d_size=16kB \
--l1d_assoc=4 \
--num-cpus=1 \
--cmd=benchmarks/x86/FFT --options="-p1 -m16"
