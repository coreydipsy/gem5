####### x86 #######
###################

##### FFT #####

build/X86/gem5.opt --outdir=m5out/powerful/x86/FFT configs/example/se.py \
--cpu-type=X86MinorCPU \
--cpu-clock=4GHz \
--cacheline_size=64 \
--caches \
--l1i_size=32kB \
--l1i_assoc=8 \
--l1d_size=64kB \
--l1d_assoc=8 \
--l2cache \
--l2_size=2MB \
--l2_assoc=16 \
--l3cache \
--l3_size=8MB \
--l3_assoc=16 \
--num-cpus=2 \
--cmd=benchmarks/x86/FFT --options="-p1 -m16"
